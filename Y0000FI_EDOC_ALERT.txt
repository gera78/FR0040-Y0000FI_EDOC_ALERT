*---------------------------------------------------------------------*
* Distribution channel - same logic as EDOC_COCKPIT
* (CL_EDOC_SOURCE_SD: channel read from VBRK with the Source Key)
*   - Source Key of a billing document = billing document number
*     (up to 10 characters) -> VBRK-VTWEG
*   - Source Key of an accounting document (FI posting) = company code
*     + document number + fiscal year (18 characters) -> no channel
*   - billing document not found in VBRK -> no channel
* Then only the eDocuments with a channel in SO_VTWEG are kept.
*---------------------------------------------------------------------*
FORM get_channel.

  TYPES: BEGIN OF ty_vbeln,
           vbeln TYPE vbeln_vf,
         END OF ty_vbeln.

  DATA lt_vbeln TYPE SORTED TABLE OF ty_vbeln WITH UNIQUE KEY vbeln.
  DATA lv_vbeln TYPE vbeln_vf.

  " Billing documents behind the eDocuments
  LOOP AT lt_edoc INTO ls_edoc.
    PERFORM source_key_to_vbeln USING ls_edoc-source_key
                                CHANGING lv_vbeln.
    IF lv_vbeln IS NOT INITIAL.
      INSERT VALUE #( vbeln = lv_vbeln ) INTO TABLE lt_vbeln.
    ENDIF.
  ENDLOOP.

  " Distribution channel from the billing document header
  IF lt_vbeln IS NOT INITIAL.
    SELECT vbeln, vtweg
      FROM vbrk
      FOR ALL ENTRIES IN @lt_vbeln
      WHERE vbeln = @lt_vbeln-vbeln
      INTO TABLE @DATA(lt_vbrk).
    SORT lt_vbrk BY vbeln.
  ENDIF.

  LOOP AT lt_edoc ASSIGNING FIELD-SYMBOL(<ls_edoc>).
    CLEAR <ls_edoc>-vtweg.
    PERFORM source_key_to_vbeln USING <ls_edoc>-source_key
                                CHANGING lv_vbeln.
    IF lv_vbeln IS INITIAL.
      CONTINUE.
    ENDIF.
    READ TABLE lt_vbrk INTO DATA(ls_vbrk)
      WITH KEY vbeln = lv_vbeln BINARY SEARCH.
    IF sy-subrc = 0.
      <ls_edoc>-vtweg = ls_vbrk-vtweg.
    ENDIF.
  ENDLOOP.

  " Filter by channel (empty value "=" -> eDocuments without channel)
  DELETE lt_edoc WHERE vtweg NOT IN so_vtweg.

ENDFORM.

*---------------------------------------------------------------------*
* Source Key -> billing document number (empty if it cannot be one)
*---------------------------------------------------------------------*
FORM source_key_to_vbeln USING    iv_source_key TYPE edocument-source_key
                         CHANGING cv_vbeln      TYPE vbeln_vf.

  CLEAR cv_vbeln.

  IF iv_source_key IS INITIAL OR strlen( iv_source_key ) > 10.
    RETURN.
  ENDIF.

  cv_vbeln = iv_source_key.

  CALL FUNCTION 'CONVERSION_EXIT_ALPHA_INPUT'
    EXPORTING
      input  = cv_vbeln
    IMPORTING
      output = cv_vbeln.

ENDFORM.

* Begin of Change : CTASK0111324
*---------------------------------------------------------------------*
* Read a standard text from SO10 (ST / TEXT / iv_name) in language
* P_LANGU and turn the SAPscript lines into plain lines:
*   '*', '/' or other format -> new line
*   '='                      -> continues the previous line, no space
*   ' '                      -> continues the previous line with a space
*   '/:' and '/*'            -> SAPscript command / comment, ignored
*---------------------------------------------------------------------*
FORM read_so10_text USING    iv_name  TYPE tdobname
                    CHANGING ct_lines TYPE string_table
                             cv_found TYPE abap_bool.

  DATA lt_tline TYPE STANDARD TABLE OF tline.
  DATA lv_text  TYPE string.

  CLEAR: ct_lines, cv_found.

  CALL FUNCTION 'READ_TEXT'
    EXPORTING
      id                      = c_text_id
      language                = p_langu
      name                    = iv_name
      object                  = c_text_obj
    TABLES
      lines                   = lt_tline
    EXCEPTIONS
      id                      = 1
      language                = 2
      name                    = 3
      not_found               = 4
      object                  = 5
      reference_check         = 6
      wrong_access_to_archive = 7
      OTHERS                  = 8.

  IF sy-subrc <> 0.
    RETURN.
  ENDIF.

  cv_found = abap_true.

  LOOP AT lt_tline INTO DATA(ls_tline).

    IF ls_tline-tdformat = '/:' OR ls_tline-tdformat = '/*'.
      CONTINUE.
    ENDIF.

    lv_text = ls_tline-tdline.

    IF ( ls_tline-tdformat = '=' OR ls_tline-tdformat = space )
       AND ct_lines IS NOT INITIAL.
      ASSIGN ct_lines[ lines( ct_lines ) ] TO FIELD-SYMBOL(<lv_last>).
      IF ls_tline-tdformat = '='.
        <lv_last> = <lv_last> && lv_text.
      ELSE.
        <lv_last> = |{ <lv_last> } { lv_text }|.
      ENDIF.
    ELSE.
      APPEND lv_text TO ct_lines.
    ENDIF.

  ENDLOOP.

ENDFORM.

FORM log_data.

  LOOP AT lt_edoc INTO ls_edoc.
    LOOP AT so_email INTO DATA(ls_email).

      " Duplicate Check
      SELECT SINGLE *
        FROM y0000fi_edocmail
        WHERE edoc_guid = @ls_edoc-edoc_guid
          AND email_to  = @ls_email-low
        INTO @DATA(ls_existing).

      IF sy-subrc = 0.
        CONTINUE.
      ENDIF.

      CLEAR ls_log.

      ls_log-mandt = sy-mandt.
      ls_log-edoc_guid = ls_edoc-edoc_guid.
      ls_log-email_to  = ls_email-low.
      ls_log-source_key = ls_edoc-source_key.

      " ✅ KORREKT: METASTATUS
      ls_log-status = ls_edoc-metastatus.

      READ TABLE lt_status_values INTO ls_status_value
        WITH KEY domvalue_l = ls_edoc-metastatus.

      IF sy-subrc = 0.
        ls_log-status_text = ls_status_value-ddtext.
      ENDIF.

      ls_log-source_created_by = ls_edoc-source_created_by.
      ls_log-sent_date = sy-datum.
      ls_log-sent_time = sy-uzeit.

      INSERT y0000fi_edocmail FROM ls_log.

    ENDLOOP.
  ENDLOOP.

ENDFORM.
