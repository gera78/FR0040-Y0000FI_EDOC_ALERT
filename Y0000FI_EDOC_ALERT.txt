REPORT y0000fi_edoc_alert.

*---------------------------------------------------------------------*
* Selection Screen
*---------------------------------------------------------------------*
SELECTION-SCREEN BEGIN OF BLOCK b1 WITH FRAME TITLE TEXT-001.

  data dummy_email  type ad_smtpadr.
  data lv_credat    type edocument-create_date.
  data lv_edoc_type type edoc_type.

  SELECT-OPTIONS: so_email FOR dummy_email NO INTERVALS OBLIGATORY.
  SELECT-OPTIONS: so_crdat FOR lv_credat.
  SELECT-OPTIONS: so_edoct FOR lv_edoc_type.
  PARAMETERS:     p_langu  TYPE spras DEFAULT sy-langu OBLIGATORY.

SELECTION-SCREEN END OF BLOCK b1.

*---------------------------------------------------------------------*
* Constants
*---------------------------------------------------------------------*
CONSTANTS:
  " Default subject - used when the SO10 text has no 'Subject:' line
  c_subject   TYPE string     VALUE 'Action Required - eDocument Transmission Errors from Previous Day',
  " First line of the SO10 text starting with this prefix is the subject
  c_subj_pref TYPE string     VALUE 'SUBJECT:',
  " Standard text (SO10) with the e-mail body
  c_text_name TYPE tdobname   VALUE 'Y0000FI_EDOC_ALERT_BODY',
  c_text_id   TYPE tdid       VALUE 'ST',
  c_text_obj  TYPE tdobject   VALUE 'TEXT',
  " Placeholder in the SO10 text replaced by the list of eDocuments
  c_ph_list   TYPE string     VALUE '&EDOC_LIST&'.

TYPES: BEGIN OF ty_edoc,
         edoc_guid         TYPE edocument-edoc_guid,
         source_key        TYPE edocument-source_key,
         source_created_by TYPE edocument-source_created_by,
         metastatus        TYPE edocument-metastatus,
         create_date       TYPE edocument-create_date,
       END OF ty_edoc.


*---------------------------------------------------------------------*
* Data Declarations
*---------------------------------------------------------------------*
DATA:
  lt_edoc          TYPE TABLE OF ty_edoc,
  ls_edoc          TYPE ty_edoc,
  email_body       TYPE STANDARD TABLE OF string,
  wa_body_line     TYPE string,

  ls_log           TYPE y0000fi_edocmail,

  lt_status_values TYPE TABLE OF dd07v,
  ls_status_value  TYPE dd07v.





*---------------------------------------------------------------------*
* Selection screen check: SO10 text must exist in the chosen language
*---------------------------------------------------------------------*
AT SELECTION-SCREEN ON p_langu.

  DATA lt_check_lines TYPE string_table.
  DATA lv_check_found TYPE abap_bool.

  PERFORM read_body_text CHANGING lt_check_lines lv_check_found.
  IF lv_check_found = abap_false.
    MESSAGE |Standard text { c_text_name } not found in language { p_langu } (SO10)| TYPE 'E'.
  ENDIF.

*---------------------------------------------------------------------*
* Start of Selection
*---------------------------------------------------------------------*
START-OF-SELECTION.

  PERFORM get_status_texts.
  PERFORM get_data.
  PERFORM send_email.
  PERFORM log_data.

  IF sy-batch <> abap_true.
    PERFORM display_alv.
  ENDIF.

*---------------------------------------------------------------------*
* Get Data
*---------------------------------------------------------------------*
FORM get_data.

  SELECT edoc_guid,
         source_key,
         source_created_by,
         metastatus,
         create_date
    FROM edocument
    INTO TABLE @lt_edoc
    WHERE edoc_type   in @so_edoct
      AND metastatus <> '0001'     " NOT Completed
      AND create_date IN @so_crdat.

ENDFORM.

*---------------------------------------------------------------------*
* Send Email
*---------------------------------------------------------------------*
FORM send_email.

  CLEAR email_body.

  IF lt_edoc IS INITIAL.
    RETURN.
  ENDIF.


  DATA lt_template TYPE string_table.
  DATA lt_list     TYPE string_table.
  DATA lv_found    TYPE abap_bool.
  DATA lv_ph_used  TYPE abap_bool.
  DATA lv_subject  TYPE string.

  " E-mail body text from SO10 (Y0000FI_EDOC_ALERT_BODY) in P_LANGU
  PERFORM read_body_text CHANGING lt_template lv_found.
  IF lv_found = abap_false.
    MESSAGE |Standard text { c_text_name } not found in language { p_langu } (SO10)| TYPE 'E'.
  ENDIF.

  " Subject: first line of the SO10 text if it starts with 'Subject:'
  " (that line and the blank lines right after it are not in the body)
  lv_subject = c_subject.
  READ TABLE lt_template INTO DATA(lv_first_line) INDEX 1.
  IF sy-subrc = 0.
    lv_first_line = condense( lv_first_line ).
    " CP is not case-sensitive: 'Subject:', 'SUBJECT:' etc.
    IF lv_first_line CP |{ c_subj_pref }*|.
      lv_subject = condense( substring( val = lv_first_line
                                        off = strlen( c_subj_pref ) ) ).
      DELETE lt_template INDEX 1.
      WHILE lt_template IS NOT INITIAL AND lt_template[ 1 ] IS INITIAL.
        DELETE lt_template INDEX 1.
      ENDWHILE.
      IF lv_subject IS INITIAL.
        lv_subject = c_subject.
      ENDIF.
    ENDIF.
  ENDIF.

  " List of eDocuments - goes where the SO10 text has &EDOC_LIST&
  LOOP AT lt_edoc INTO ls_edoc.

    APPEND |eDocument: { ls_edoc-edoc_guid }| TO lt_list.

    " Get readable status text
    READ TABLE lt_status_values INTO ls_status_value
      WITH KEY domvalue_l = ls_edoc-metastatus.

    IF sy-subrc = 0.
      APPEND |Status: { ls_status_value-ddtext }| TO lt_list.
    ELSE.
      APPEND |Status: { ls_edoc-metastatus }| TO lt_list.
    ENDIF.

    APPEND |Created by: { ls_edoc-source_created_by }| TO lt_list.
    APPEND |Creation Date: { ls_edoc-create_date }| TO lt_list.
    APPEND |Source Key: { ls_edoc-source_key }| TO lt_list.
    APPEND |----------------------------------------| TO lt_list.

  ENDLOOP.

  " Merge: SO10 text with the list in place of the placeholder line
  LOOP AT lt_template INTO DATA(lv_template_line).
    IF lv_template_line CS c_ph_list.
      APPEND LINES OF lt_list TO email_body.
      lv_ph_used = abap_true.
    ELSE.
      APPEND lv_template_line TO email_body.
    ENDIF.
  ENDLOOP.

  " Placeholder missing in the SO10 text: list goes at the end
  IF lv_ph_used = abap_false.
    APPEND | | TO email_body.
    APPEND LINES OF lt_list TO email_body.
  ENDIF.

  DATA(lo_message) = NEW cl_bcs_message( ).
  lo_message->set_subject( lv_subject ).

  CONCATENATE LINES OF email_body INTO wa_body_line
    SEPARATED BY cl_abap_char_utilities=>cr_lf.

  lo_message->set_main_doc( wa_body_line ).

  LOOP AT so_email INTO DATA(ls_email).
    lo_message->add_recipient(
      iv_address = CONV string( ls_email-low ) ).
  ENDLOOP.

  lo_message->set_send_immediately( abap_true ).
  lo_message->set_update_task( abap_false ).

  TRY.
      lo_message->send( ).
    CATCH cx_bcs_send INTO DATA(lx_send).
      MESSAGE lx_send->get_text( ) TYPE 'E'.
  ENDTRY.

ENDFORM.

*---------------------------------------------------------------------*
* Read the e-mail body from SO10 (ST / TEXT / Y0000FI_EDOC_ALERT_BODY)
* in language P_LANGU and turn the SAPscript lines into plain lines:
*   '*', '/' or other format -> new line
*   '='                      -> continues the previous line, no space
*   ' '                      -> continues the previous line with a space
*   '/:' and '/*'            -> SAPscript command / comment, ignored
*---------------------------------------------------------------------*
FORM read_body_text CHANGING ct_lines TYPE string_table
                             cv_found TYPE abap_bool.

  DATA lt_tline TYPE STANDARD TABLE OF tline.
  DATA lv_text  TYPE string.

  CLEAR: ct_lines, cv_found.

  CALL FUNCTION 'READ_TEXT'
    EXPORTING
      id                      = c_text_id
      language                = p_langu
      name                    = c_text_name
      object                  = c_text_obj
    TABLES
      lines                   = lt_tline
    EXCEPTIONS
      id                      = 1
      language                = 2
      name                    = 3
      not_found               = 4
      object                  = 5
      reference_check         = 6
      wrong_access_to_archive = 7
      OTHERS                  = 8.

  IF sy-subrc <> 0.
    RETURN.
  ENDIF.

  cv_found = abap_true.

  LOOP AT lt_tline INTO DATA(ls_tline).

    IF ls_tline-tdformat = '/:' OR ls_tline-tdformat = '/*'.
      CONTINUE.
    ENDIF.

    lv_text = ls_tline-tdline.

    IF ( ls_tline-tdformat = '=' OR ls_tline-tdformat = space )
       AND ct_lines IS NOT INITIAL.
      ASSIGN ct_lines[ lines( ct_lines ) ] TO FIELD-SYMBOL(<lv_last>).
      IF ls_tline-tdformat = '='.
        <lv_last> = <lv_last> && lv_text.
      ELSE.
        <lv_last> = |{ <lv_last> } { lv_text }|.
      ENDIF.
    ELSE.
      APPEND lv_text TO ct_lines.
    ENDIF.

  ENDLOOP.

ENDFORM.

*---------------------------------------------------------------------*
* Get Status Texts (Domain)
*---------------------------------------------------------------------*
FORM get_status_texts.

  CALL FUNCTION 'DD_DOMVALUES_GET'
    EXPORTING
      domname   = 'EDOC_METASTATUS'
      text      = 'X'
    TABLES
      dd07v_tab = lt_status_values.

ENDFORM.

*---------------------------------------------------------------------*
* Logging
*---------------------------------------------------------------------*
FORM log_data.

  LOOP AT lt_edoc INTO ls_edoc.
    LOOP AT so_email INTO DATA(ls_email).

      " Duplicate Check
      SELECT SINGLE *
        FROM y0000fi_edocmail
        WHERE edoc_guid = @ls_edoc-edoc_guid
          AND email_to  = @ls_email-low
        INTO @DATA(ls_existing).

      IF sy-subrc = 0.
        CONTINUE.
      ENDIF.

      CLEAR ls_log.

      ls_log-mandt = sy-mandt.
      ls_log-edoc_guid = ls_edoc-edoc_guid.
      ls_log-email_to  = ls_email-low.
      ls_log-source_key = ls_edoc-source_key.

      " ✅ KORREKT: METASTATUS
      ls_log-status = ls_edoc-metastatus.

      READ TABLE lt_status_values INTO ls_status_value
        WITH KEY domvalue_l = ls_edoc-metastatus.

      IF sy-subrc = 0.
        ls_log-status_text = ls_status_value-ddtext.
      ENDIF.

      ls_log-source_created_by = ls_edoc-source_created_by.
      ls_log-sent_date = sy-datum.
      ls_log-sent_time = sy-uzeit.

      INSERT y0000fi_edocmail FROM ls_log.

    ENDLOOP.
  ENDLOOP.

ENDFORM.

*---------------------------------------------------------------------*
* ALV Display
*---------------------------------------------------------------------*
FORM display_alv.

  TRY.
      cl_salv_table=>factory(
        IMPORTING r_salv_table = DATA(lo_alv)
        CHANGING  t_table      = lt_edoc ).

      lo_alv->display( ).

    CATCH cx_salv_msg.
  ENDTRY.

ENDFORM.
